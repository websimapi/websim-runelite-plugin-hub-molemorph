MoleMorphPlugin.java

// MoleMorphPlugin.java
package net.runelite.client.plugins.molemorph;

import com.google.inject.Provides;
import javax.inject.Inject;
import net.runelite.api.*;
import net.runelite.api.events.GameTick;
import net.runelite.client.config.ConfigManager;
import net.runelite.client.eventbus.Subscribe;
import net.runelite.client.plugins.Plugin;
import net.runelite.client.plugins.PluginDescriptor;

@PluginDescriptor(
    name = "MoleMorph",
    description = "Transforms your character into the Baby Mole pet!",
    tags = {"mole", "transform", "visual", "npc", "pet"}
)
public class MoleMorphPlugin extends Plugin {

    @Inject
    private Client client;

    @Inject
    private MoleMorphConfig config;

    private final int MOLE_NPC_ID = NpcID.BABY_MOLE;
    private final int INVALID_TRANSFORM_ID = -1;

    @Provides
    MoleMorphConfig provideConfig(ConfigManager configManager) {
        return configManager.getConfig(MoleMorphConfig.class);
    }

    @Override
    protected void startUp() throws Exception {
        updatePlayerModel();
    }

    @Override
    protected void shutDown() throws Exception {
        resetPlayerModel();
    }

    @Subscribe
    public void onGameTick(GameTick gameTick) {
        updatePlayerModel();
    }

    private void updatePlayerModel() {
        Player localPlayer = client.getLocalPlayer();
        if (localPlayer == null) {
            return;
        }

        // If the plugin is enabled in the config, transform the player.
        // Otherwise, change them back.
        if (config.enableMoleMorph()) {
            localPlayer.getPlayerComposition().setTransmogId(MOLE_NPC_ID);
        } else {
            // Check if we are currently transformed before resetting
            if (localPlayer.getPlayerComposition().getTransmogId() == MOLE_NPC_ID) {
                resetPlayerModel();
            }
        }
    }

    private void resetPlayerModel() {
        Player localPlayer = client.getLocalPlayer();
        if (localPlayer != null) {
            // Setting the transmog ID to -1 reverts the player to their original appearance.
            localPlayer.getPlayerComposition().setTransmogId(INVALID_TRANSFORM_ID);
        }
    }
}

MoleMorphConfig.java

// MoleMorphConfig.java
package net.runelite.client.plugins.molemorph;

import net.runelite.client.config.Config;
import net.runelite.client.config.ConfigGroup;
import net.runelite.client.config.ConfigItem;

@ConfigGroup("molemorph")
public interface MoleMorphConfig extends Config {

    @ConfigItem(
        keyName = "enableMoleMorph",
        name = "Enable Mole Morph",
        description = "Toggles the character transformation on or off.",
        position = 1
    )
    default boolean enableMoleMorph() {
        return true;
    }
}
